#3.Adding trunks 802.3ad_Trunk01 & 802.3ad_Trunk02 & assigning
#  them to the 'x' interfaces
#---------------------------------------------------------------------
        edit "802.3ad_Trunk01"
            set vdom "root"
            set type aggregate
            set member "x3" "x1"
            set description ""
            set alias "ACI Border Link"
            set device-identification enable
            set lldp-reception enable
            set lldp-transmission enable
            unset role
            set snmp-index 140
            set ip-managed-by-fortiipam disable
        next
        edit "802.3ad_Trunk02"
            set vdom "root"
            set type aggregate
            set member "x4" "x2"
            set alias "Fusion Router Link"
            set device-identification enable
            set lldp-reception enable
            set lldp-transmission enable
            unset role
            set snmp-index 141
            set ip-managed-by-fortiipam disable
        next
#---------------------------------------------------------------------
#4.Updating the managment interface using variable $(ship_IP)
#---------------------------------------------------------------------
        edit "mgmt"
      set vdom "root"
      set mode static
      set ip 10.$(ship_IP).0.47 255.255.255.0
      set allowaccess ping https ssh snmp
      set trust-ip-1 10.0.0.0/8
      set dedicated-to management
      set lldp-reception enable
      set lldp-transmission enable
      unset role
      set snmp-index 131
        next
#---------------------------------------------------------------------
#5.Creating VLAN_900 & updating port8(Legacy VLAN_112) using 
#  variable $(ship_IP)
#---------------------------------------------------------------------
        edit "VLAN_900"
            set vdom "root"
            set ip 10.$(ship_IP).254.10 255.255.255.248
            set allowaccess https ping ssh snmp http fgfm ftm fabric
            set device-identification enable
            unset role
            set snmp-index 142
            set interface "802.3ad_Trunk02"
            set vlanid 900
            set ip-managed-by-fortiipam disable
        next
        edit "port8"
            set vdom "root"
            set mode static
            set ip 10.$(ship_IP).254.241 255.255.255.240
            set allowaccess ping https ssh http fgfm ftm

        next
